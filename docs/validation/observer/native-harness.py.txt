def label(node):return node.get('text') or node.get('content-desc') or ''
def match(text,name):return text==name or text.startswith(name+'\n') or text.startswith(name+':')
def tap_control(name,scroll=False):
    for attempt in range(6 if scroll else 3):
        nodes=dump()
        for node in nodes:
            if match(label(node),name):
                x1,y1,x2,y2=map(int,re.findall(r'\d+',node.get('bounds')))
                if x2>x1 and y2>y1 and 20<(y1+y2)//2<610:
                    point=((x1+x2)//2,(y1+y2)//2)
                    adb('shell','input','tap',str(point[0]),str(point[1]))
                    print('Tap: '+name,flush=True)
                    return point
        if scroll:
            boxes=[tuple(map(int,re.findall(r'\d+',n.get('bounds')))) for n in nodes if n.get('scrollable')=='true']
            x1,y1,x2,y2=max(boxes,key=lambda b:(b[2]-b[0])*(b[3]-b[1])) if boxes else (20,130,300,610)
            adb('shell','input','swipe',str((x1+x2)//2),str(min(575,y2-40)),str((x1+x2)//2),str(max(y1+40,165)),'350')
        else:time.sleep(.5)
    raise RuntimeError('Control not visible: '+name+';'+repr([label(n) for n in nodes if label(n)]))

def wait_world(predicate,timeout=180):
    deadline=time.monotonic()+timeout
    while time.monotonic()<deadline:
        w=world()
        if predicate(w):return w
        time.sleep(1)
    raise RuntimeError('Native world did not reach the required state')

def map_capture(name):
    colors=[(126,148,96),(64,91,71),(182,156,108),(118,120,117),
            (140,155,135),(66,108,117),(130,147,101),(101,126,88),
            (133,136,123),(116,140,119),(108,133,128)]
    deadline=time.monotonic()+90
    while time.monotonic()<deadline:
        raw=adb('exec-out','screencap','-p')
        im=Image.open(io.BytesIO(raw)).convert('RGB').crop((8,120,260,350))
        count=sum(n for n,c in im.getcolors(100000)
                  if any(max(abs(c[i]-g[i]) for i in range(3))<=12 for g in colors))
        if count>6000:
            (out/name).write_bytes(raw);print('Rendered: '+name,flush=True);return
        time.sleep(.5)
    raise RuntimeError('No actual terrain; refusing premature capture')
