import hashlib, json, pathlib, re, subprocess, time

root = pathlib.Path('/workspace/Barrio_Cero')
out = root / 'docs/validation/experience/native'
out.mkdir(parents=True, exist_ok=True)
pkg = 'com.barriocero.eras'
folder = 'app_flutter/barrio_cero'
before = 'app_flutter/native_before_13_experience_retry'
after = 'app_flutter/native_after_13_experience_retry'
apk = root / 'build/deliverables/barrio-cero-eras-0.13.0-profile.apk'
result = {'version': '0.13.0+14', 'rulesVersion': 11,
          'device': 'Android API29 software emulator, 320x640',
          'scope': 'Fresh prehistory UI, optional guide, quoted building, completion, navigation and exact reopening. Not full first-ten or reference-phone acceptance.'}

def adb(*args):
    return subprocess.check_output(['adb', *args], timeout=120)

def run(*args):
    return adb('shell', 'run-as', pkg, *args)

def exists(path):
    return subprocess.run(['adb', 'shell', 'run-as', pkg, 'ls', '-d', path], capture_output=True, timeout=40).returncode == 0

def stop():
    adb('shell', 'am', 'force-stop', pkg)
    deadline = time.monotonic() + 30
    while subprocess.run(['adb', 'shell', 'pidof', pkg], capture_output=True, timeout=10).stdout.strip():
        if time.monotonic() > deadline:
            raise RuntimeError('Application did not stop')
        time.sleep(.2)

def inventory():
    result = {}
    for name in run('ls', '-1', folder).decode().splitlines():
        if not re.fullmatch(r'[A-Za-z0-9_.-]+', name):
            raise RuntimeError('Unexpected save filename')
        raw = adb('exec-out', 'run-as', pkg, 'cat', folder + '/' + name)
        result[name] = {'bytes': len(raw), 'sha256': hashlib.sha256(raw).hexdigest()}
    return result

def dump():
    for _ in range(8):
        response = adb('shell', 'uiautomator', 'dump', '/sdcard/barrio-experience.xml')
        if b'dumped to:' not in response:
            continue
        raw = adb('exec-out', 'cat', '/sdcard/barrio-experience.xml')
        import xml.etree.ElementTree as ET
        (out / 'last-window.xml').write_bytes(raw)
        return list(ET.fromstring(raw).iter('node'))
    raise RuntimeError('No fresh Android UI hierarchy')

def tap(label, scroll=False):
    for attempt in range(5 if scroll else 2):
        nodes = dump()
        for node in nodes:
            text = node.get('text') or node.get('content-desc') or ''
            if text == label or text.startswith(label + '\n'):
                x1, y1, x2, y2 = map(int, re.findall(r'\d+', node.get('bounds')))
                x, y = (x1 + x2) // 2, (y1 + y2) // 2
                if x2 > x1 and y2 > y1 and 20 <= y < 610:
                    adb('shell', 'input', 'tap', str(x), str(y))
                    print('Tap: ' + label, flush=True)
                    return
        if scroll:
            adb('shell', 'input', 'swipe', '252', '481', '252', '350', '350')
        else:
            time.sleep(.5)
    raise RuntimeError('Missing control ' + label + ': ' + repr([(n.get('text') or n.get('content-desc')) for n in nodes if n.get('text') or n.get('content-desc')]))

def capture(name):
    time.sleep(.5)
    (out / name).write_bytes(adb('exec-out', 'screencap', '-p'))

def rawworld():
    return adb('exec-out', 'run-as', pkg, 'cat', folder + '/world_0.json')

def world():
    return json.loads(rawworld())['payload']['world']

def waitworld(predicate):
    deadline = time.monotonic() + 45
    while time.monotonic() < deadline:
        try:
            value = world()
            if predicate(value):
                return value
        except (subprocess.CalledProcessError, json.JSONDecodeError):
            pass
        time.sleep(.25)
    raise RuntimeError('World never reached the required confirmed state')

moved = False
try:
    assert apk.is_file()
    stop()
    if exists(before) or exists(after):
        raise RuntimeError('Retained native fixture exists; refusing overwrite')
    original = inventory()
    (out / 'original-inventory.json').write_text(json.dumps(original, indent=2) + '\n')
    result['installationOutput'] = adb('install', '-r', str(apk)).decode()
    result['apkSha256'] = hashlib.sha256(apk.read_bytes()).hexdigest()
    run('mv', folder, before)
    moved = True
    run('mkdir', folder)
    adb('shell', 'am', 'start', '-n', pkg + '/.MainActivity')
    tap('CREAR UN MUNDO')
    tap('Fundar Barrio Cero')
    initial = waitworld(lambda w: w['day'] == 0 and w['origin'] == 'prehistory')
    source = rawworld()
    result['freshOrigin'] = initial['origin']
    result['seed'] = initial['seed']
    capture('01-real-prehistory-needs.png')
    tap('Guía opcional', scroll=True)
    assert rawworld() == source
    capture('02-real-optional-guide.png')
    tap('Seguir libremente')
    tap('Ampliar refugios', scroll=True)
    assert rawworld() == source
    capture('03-real-cost-before-confirming.png')
    tap('Confirmar orden')
    ordered = waitworld(lambda w: len(w['societies'][0]['projects']) == 1)
    assert ordered['societies'][0]['resources']['wood'] == 22
    assert ordered['societies'][0]['resources']['stone'] == 14
    assert ordered['rng'] == initial['rng']
    result['quotePreservedCheckpoint'] = True
    result['actualBuildingCost'] = {'wood': 8, 'stone': 2}
    tap('Avanzar')
    tap('Avanzar un mes')
    completed = waitworld(lambda w: w['day'] >= 10 and not w['societies'][0]['projects'])
    assert completed['societies'][0]['built']['shelter'] == 4
    assert len(completed['societies'][0]['sites']) == len(initial['societies'][0]['sites']) + 1
    final = rawworld()
    result['persistentCompletion'] = {'day': completed['day'], 'shelters': 4, 'actualSiteAdded': True}
    capture('04-real-completed-refuge.png')
    tap('Guía opcional', scroll=True)
    nodes = dump()
    assert any('1 de 10 decisiones registradas' in (n.get('text') or n.get('content-desc') or '') for n in nodes)
    capture('05-real-confirmed-decision.png')
    tap('Seguir libremente')
    tap('Mundo')
    capture('06-real-world.png')
    tap('Historia')
    capture('07-real-history.png')
    tap('Barrio')
    assert rawworld() == final
    stop()
    adb('shell', 'am', 'start', '-n', pkg + '/.MainActivity')
    tap('Continuar historia')
    dump()
    assert rawworld() == final
    result['forceStopReopenExact'] = True
    result['navigationPreservedWorld'] = True
    capture('08-real-reopened.png')
except Exception as error:
    result['error'] = repr(error)
    print(repr(error), flush=True)
finally:
    if moved:
        stop()
        if exists(after):
            raise RuntimeError('Refusing to overwrite retained native fixture')
        if exists(folder):
            run('mv', folder, after)
        if exists(folder):
            raise RuntimeError('Original restoration destination is not empty')
        run('mv', before, folder)
        result['originalSavesRestoredExactly'] = inventory() == original
        if not result['originalSavesRestoredExactly']:
            raise RuntimeError('Original saves mismatch')
    result['passed'] = 'error' not in result
    (out / 'results.json').write_text(json.dumps(result, ensure_ascii=False, indent=2) + '\n')
    print(json.dumps(result, ensure_ascii=False), flush=True)
if not result['passed']:
    raise SystemExit(1)
