"""Collect already-running tests; never implement or approve unexecuted work."""
import datetime
import json
import pathlib
import shutil
import time

root = pathlib.Path('/workspace/Barrio_Cero')
out = root / 'docs/validation/transit'
jobs = {
    'opening': (130209, 'transit-r2-opening-benchmark.log', 'rules11-r2-opening-benchmark'),
    'mature': (130184, 'transit-r2-mature-benchmark.log', 'rules11-r2-mature-benchmark'),
    'stability': (130213, 'transit-r2-stability.log', 'rules11-r2-simulation-check'),
}

def rows(text):
    found = []
    for line in text.splitlines():
        if not line.startswith('{'):
            continue
        try:
            found.append(json.loads(line))
        except json.JSONDecodeError:
            pass  # A concurrent write has not finished its last line yet.
    return found

def alive(pid, name):
    try:
        return name.encode() in pathlib.Path(f'/proc/{pid}/cmdline').read_bytes()
    except FileNotFoundError:
        return False

deadline = time.monotonic() + 12 * 3600
while True:
    report = json.loads((out / 'results.json').read_text())
    complete = True
    for name, (pid, source, executable) in jobs.items():
        path = root / 'build/validation' / source
        text = path.read_text()
        records = rows(text)
        running = alive(pid, executable)
        shutil.copyfile(path, out / source.removeprefix('transit-r2-'))
        if name in ('opening', 'mature'):
            key = 'sustained32'
            report[key][name] = records
            done = len(records) == 2
            report[key][name + 'State'] = 'complete' if done else 'in_progress' if running else 'incomplete'
            flag = 'sustained32SettlementScenario' if name == 'opening' else 'passedSustainedLoad'
            report[key][name + 'Accepted'] = done and all(r[flag] for r in records)
            complete = complete and not running
        else:
            passed = len(records) == 32 and 'PASS full-content 30 seeds' in text
            report['stability'] = {
                'state': 'complete' if passed else 'in_progress' if running else 'incomplete',
                'completedScenarios': records,
                'fullMatrixPassed': passed,
                'activeMillenniaVerified': passed and all(
                    r['actualDays'] == 360000 and r['terminal'] == 'target_reached'
                    for r in records if r['requestedDays'] == 360000
                ),
            }
            complete = complete and not running
    cases = report['sustained32']
    cases['state'] = 'complete' if all(cases.get(k + 'State') == 'complete' for k in ('opening', 'mature')) else 'in_progress'
    report['recordedAt'] = datetime.datetime.now(datetime.timezone.utc).isoformat()
    temp = out / 'results.collecting.json'
    temp.write_text(json.dumps(report, ensure_ascii=False, indent=2) + '\n')
    temp.replace(out / 'results.json')
    if complete or time.monotonic() >= deadline:
        print(json.dumps({'collectorFinished': True, 'allJobsFinished': complete, 'stability': report['stability']['state'], 'sustained32': cases['state']}), flush=True)
        break
    time.sleep(30)
